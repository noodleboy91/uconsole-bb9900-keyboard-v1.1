# Run these on the uConsole over SSH or using an external keyboard.
# On stock firmware: hold Right Fn and press backslash (\) to enter bootloader.

# Confirm /dev/sda is the keyboard bootloader drive before mounting.
# The user's V1.1 drive was labelled NRF52BOOT; device names can change.
lsblk -o NAME,LABEL,SIZE,FSTYPE,MOUNTPOINTS

# Mount the bootloader drive (skip mount if already mounted here):
sudo mkdir -p /mnt/keyboard
sudo mount /dev/sda /mnt/keyboard

# Back up the current firmware without overwriting an existing backup:
cp -n /mnt/keyboard/CURRENT.UF2 ~/keyboard-backup.uf2

# Download firmware from the latest release:
mkdir -p ~/Downloads

# CHOOSE ONE firmware below. Download it first, then copy it.

# LEFT CLICK: pressing the trackpad sends left mouse click.
curl -fL 'https://github.com/noodleboy91/uconsole-bb9900-keyboard-v1.1/releases/download/v1.5.0-v1.1/bb9900-v1.1-zmk.uf2' \
  -o ~/Downloads/bb9900-v1.1-zmk.uf2
sudo cp ~/Downloads/bb9900-v1.1-zmk.uf2 /mnt/keyboard/
sync
